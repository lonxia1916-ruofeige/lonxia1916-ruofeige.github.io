    <script>
        // 登录方式切换
        const tabs = document.querySelectorAll('.tab-item');
        const panels = document.querySelectorAll('.form-panel');

        tabs.forEach(tab => {
            tab.addEventListener('click', () => {
                const target = tab.dataset.tab;
                
                // 切换tab状态
                tabs.forEach(t => t.classList.remove('active'));
                tab.classList.add('active');
                
                // 切换面板
                panels.forEach(p => p.classList.remove('active'));
                document.getElementById(target + '-panel').classList.add('active');
            });
        });

        // 获取验证码倒计时
        const codeBtn = document.getElementById('getCodeBtn');
        let countdown = 60;
        let timer = null;

        codeBtn.addEventListener('click', () => {
            if (timer) return;
            
            codeBtn.disabled = true;
            codeBtn.textContent = `${countdown}s后重发`;
            
            timer = setInterval(() => {
                countdown--;
                codeBtn.textContent = `${countdown}s后重发`;
                
                if (countdown <= 0) {
                    clearInterval(timer);
                    timer = null;
                    countdown = 60;
                    codeBtn.disabled = false;
                    codeBtn.textContent = '获取验证码';
                }
            }, 1000);
        });

        // 保存至桌面（创建快捷方式提示）
        function saveToDesktop() {
            alert('请按 Ctrl+D（Mac：Command+D）将页面保存到书签，或手动创建桌面快捷方式');
        }

        // 添加至收藏夹
        function addFavorite() {
            const title = document.title;
            const url = window.location.href;
            
            try {
                // IE
                if (window.external && window.external.AddFavorite) {
                    window.external.AddFavorite(url, title);
                }
                // 其他浏览器提示
                else {
                    alert('请按 Ctrl+D（Mac：Command+D）添加到收藏夹');
                }
            } catch (e) {
                alert('请按 Ctrl+D（Mac：Command+D）添加到收藏夹');
            }
        }

        // 登录按钮示例
        document.querySelector('.login-btn').addEventListener('click', () => {
            alert('登录功能需对接后端接口，此处为演示效果');
        });
    </script>
